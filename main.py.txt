conn = sqlite3.connect("dealhunter.db", check_same_thread=False)
conn.execute("CREATE TABLE IF NOT EXISTS prices (model TEXT, price REAL, checked_at TEXT)")
conn.execute("CREATE TABLE IF NOT EXISTS alerts (model TEXT, drop_pct REAL, sent_at TEXT)")
conn.commit()

def get_reference(model):
    cur = conn.execute("SELECT price FROM prices WHERE model=? ORDER BY checked_at DESC LIMIT 60", (model,))
    prices = [r[0] for r in cur.fetchall()]
    if len(prices) < 5:
        return None
    s = sorted(prices)
    return statistics.median(s[:int(len(s)*0.9)])

def format_alert(cat, p, ref, cur_price, drop, save):
    amz = f"https://www.amazon.in/s?k={p['search'].replace(' ', '+')}"
    fk = f"https://www.flipkart.com/search?q={p['search'].replace(' ', '%20')}"
    ts = datetime.now().strftime("%d-%m-%Y %I:%M %p")
    if cat == "GOOD":
        return f"🟢 GOOD DEAL FOUND!\n\nProduct: {p['name']}\nBrand: {p['brand']}\nPrevious Reference Price: ₹{ref:,.0f}\nCurrent Price: ₹{cur_price:,.0f}\nPrice Drop: {drop:.0f}%\nYou Save: ₹{save:,.0f}\n\nAmazon: {amz}\nFlipkart: {fk}\nLast Checked: {ts}"
    if cat == "EXCELLENT":
        return f"🔥 EXCELLENT PRICE DROP!\n\nProduct: {p['name']}\nBrand: {p['brand']}\nReference Price: ₹{ref:,.0f}\nCurrent Price: ₹{cur_price:,.0f}\nDiscount: {drop:.0f}%\nYou Save: ₹{save:,.0f}\n\nAmazon: {amz}\nFlipkart: {fk}"
    return f"🚨🚨 MASSIVE PRICE DROP ALERT 🚨🚨\n\n⚡ {drop:.0f}%+ PRICE DROP DETECTED!\n\nProduct: {p['name']}\nBrand: {p['brand']}\nHistorical Reference Price: ₹{ref:,.0f}\nCurrent Price: ₹{cur_price:,.0f}\nPrice Drop: {drop:.0f}%\nYou Save: ₹{save:,.0f}\n\n🔥 THIS IS A HIGH-PRIORITY DEAL!\n🛒 Amazon: {amz}\n🛒 Flipkart: {fk}\nDetection Time: {ts}"

bot = Bot(token=BOT_TOKEN)
dp = Dispatcher()
sched = AsyncIOScheduler()

@dp.message(Command("start"))
async def start(msg: Message):
    if msg.from_user.id != ALLOWED_ID: return
    await msg.answer(f"🤖 Laptop Deal Hunter Started!\n\nHP + Lenovo + ASUS + Acer\n68 Models Tracked\nFilter: i7/Ryzen7 and ABOVE + 1TB SSD+ + Best Config\nCheck: Every 15 mins\n\n/status - Status\n/tracked - List\n/deals - Recent Deals\n/massive - Only 60%+")

@dp.message(Command("status"))
async def status(msg: Message):
    if msg.from_user.id != ALLOWED_ID: return
    await msg.answer(f"🟢 Bot Running\nTracked: {len(PRODUCTS)} Models\nBrands: HP, Lenovo, ASUS, Acer\nFilter: i7+ / 1TB+ / Best Config\nCheck: Every 15 mins")

@dp.message(Command("tracked"))
async def tracked(msg: Message):
    if msg.from_user.id != ALLOWED_ID: return
    txt = "📦 Tracked 68 Laptops:\n\n"
    for p in PRODUCTS:
        txt += f"- {p['brand']} | {p['name'][:65]}\n"
    # Telegram limit 4096, split
    for i in range(0, len(txt), 4000):
        await msg.answer(txt[i:i+4000])

@dp.message(Command("deals"))
async def deals(msg: Message):
    if msg.from_user.id != ALLOWED_ID: return
    cur = conn.execute("SELECT model, drop_pct, sent_at FROM alerts ORDER BY sent_at DESC LIMIT 10")
    rows = cur.fetchall()
    if not rows:
        await msg.answer("No deals yet. Bot is collecting history - needs 5 checks.")
        return
    await msg.answer("\n".join([f"{r[0]} - {r[1]:.0f}% - {r[2]}" for r in rows]))

@dp.message(Command("massive"))
async def massive(msg: Message):
    if msg.from_user.id != ALLOWED_ID: return
    cur = conn.execute("SELECT model, drop_pct FROM alerts WHERE drop_pct>=60 ORDER BY sent_at DESC LIMIT 10")
    rows = cur.fetchall()
    await msg.answer(f"🚨 Massive Deals: {len(rows)}" if rows else "No massive deals yet.")

@dp.message(Command("help"))
async def help_cmd(msg: Message):
    await msg.answer("/start /status /tracked /deals /massive /help")

async def check_prices():
    print(f"[{datetime.now()}] Checking {len(PRODUCTS)} models...")

async def main():
    sched.add_job(check_prices, "interval", minutes=15)
    sched.start()
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())
